//@version=5
indicator('AO Vwap SuperTrend', shorttitle='AO', overlay=true)

fastLength = 5  //input(title="Fast Length", type=input.integer, defval=5)
slowLength = 34  //input(title="Slow Length", type=input.integer, defval=34)

ao = ta.sma(hl2, fastLength) - ta.sma(hl2, slowLength)

aoColor = ao >= 0 ? ao[1] < ao ? #26A69A : #B2DFDB : ao[1] < ao ? #FFCDD2 : #EF5350
plot(ao, title='AO', style=plot.style_columns, color=aoColor, transp=0)

//plotshape(ao[1] < ao , color = color.green , style = shape.labelup , text = "Buy" , textcolor = color.white , location = location.belowbar) 
//plotshape(ao[1] > ao   , color = color.red , style = shape.labeldown , text = "Sell" , textcolor = color.white , location = location.abovebar) 

VWAP = ta.vwap(hlc3)
//plot(VWAP , color = color.red , linewidth = 2) 
buya = close > VWAP  //and uptrend
sella = close < VWAP  //and dntrend

//inputs 
Periods = input(title='p', defval=22)
Source = input(close, title='Source')
Multiplier = input.float(title='m', step=0.1, defval=5.0)



//Compute ATR Levels 
atr = ta.atr(Periods)


//Creating Upper Channel 

up = Source - Multiplier * atr
up1 = nz(up[1], up)
up := close[1] > up1 ? math.max(up, up1) : up

//Creating Down Channel 
dn = Source + Multiplier * atr
dn1 = nz(dn[1], dn)
dn := close[1] < dn1 ? math.min(dn, dn1) : dn


//Compute the Trend Stream +1/-1 
trend = 1
trend := nz(trend[1], trend)
trend := trend == -1 and close > dn1 ? 1 : trend == 1 and close < up1 ? -1 : trend

//Create Stoploss for Longs 
upPlot = trend == 1 ? up : na  //, title="Up Trend", style=plot.style_linebr, linewidth=2, color=color.green) 
//buy_a = close > upPlot 
//Buy Signal 
buy_3 = trend == 1 and trend[1] == -1

//plotshape(buy_3 ? up : na, title="Go Long", location=location.absolute, style=shape.circle, size=size.tiny, color=color.green, transp=0) 

dnPlot = trend == 1 ? na : dn  //, title="Down Trend", style=plot.style_linebr, linewidth=2, color=color.red) 
//sell_a = close < dnPlot 
//Sell Signal 
sell_3 = trend == -1 and trend[1] == 1

//plotshape(sell_3 ? dn : na, title="Go Short", location=location.absolute, style=shape.circle, size=size.tiny, color=color.red, transp=0)

buy_1 = buya and trend == 1 //and ao[1] < ao
sell_1 = sella and trend == -1 //and ao[1] > ao

InSession(sess) =>
    na(time(timeframe.period, sess)) == false
sessionHours = input.session(defval="0915-1500", title="Session Hours")
weekendsStr = ":1234567" 
bgcolor(InSession(sessionHours + weekendsStr)[1] ? color.red : na, title="Session Hours", transp=85)

sec=InSession(sessionHours + weekendsStr)[1]

buffer = input(0.00)/100
 
long_last1 = false
short_last1 = false
 
long_last1 := nz(long_last1[1])
short_last1 := nz(short_last1[1])
 
 
buy1 = buy_1 and not buy_1[1]//Signal
sell1 = sell_1 and not sell_1[1]//Signal
 
 
long1 = false
long1 := nz(long1[1])
 
short1 = false
short1 := nz(short1[1])
 
long_wait1 = false
long_wait1 := nz(long_wait1[1])
 
short_wait1 = false
short_wait1 := nz(short_wait1[1])
 
low_forming = sell1 //and not short_last1
high_forming = buy1 //and not long_last1
 
entry_p = 0.0
entry_p := nz(entry_p[1])
 
if(low_forming)
    entry_p := low * (1-buffer)
    short_wait1 := true
    long_wait1 := false
 
if(high_forming)
    entry_p := high * (1+buffer)
    long_wait1 := true
    short_wait1 := false
 
buy = long_wait1 and ta.crossover(high , entry_p) and  sec
sell = short_wait1 and ta.crossunder(low , entry_p) and  sec
 
 
if(buy)
    long1 := true
    long_wait1 := false
    short1 := false
    long_last1 := true
    short_last1 := false
 
if(sell)
    short1 := true
    short_wait1 := false
    long1 := false
    long_last1 := false
    short_last1 := true
 
plot(long_wait1 ? entry_p : na , color = color.green , linewidth = 2 , style = plot.style_linebr)
plot(short_wait1 ? entry_p : na , color = color.red , linewidth = 2 , style = plot.style_linebr)
 
use_sl = input(false,title = "Use Candle High/Low as SL")
 
Stop = input(1.0,title = "StopLoss")/100
 
ProfitPerc = input(defval = 1.0,title = "TARGET")/100
 
 
long_short = 0
long_last = (buy) and (nz(long_short[1]) == 0 or nz(long_short[1]) == -1)
short_last = (sell) and (nz(long_short[1]) == 0 or nz(long_short[1]) == 1)
long_short := long_last ? 1 : short_last ? -1 : long_short[1]
 
long_entered = false
long_entered := long_entered[1]
 
short_entered = false
short_entered := short_entered[1]
 
 
longPrice = ta.valuewhen(buy and not(long_entered), entry_p, 0)


shortPrice = ta.valuewhen(sell and not(short_entered), entry_p, 0)
 
if(use_sl)
    longPrice := ta.valuewhen(buy and not(long_entered),entry_p , 0)
    shortPrice := ta.valuewhen(sell and not(short_entered),entry_p, 0)
 
 
longStop = use_sl ? longPrice : longPrice * (1 - Stop)
shortStop = use_sl ? shortPrice : shortPrice * (1 + Stop)
 
if(use_sl)
    longStop := ta.valuewhen(buy and not(long_entered), low , 0)
    shortStop := ta.valuewhen(sell and not(short_entered),high, 0)
 
longTake = longPrice * (1 +  ProfitPerc)
shortTake = shortPrice* (1  - ProfitPerc)
 
plot(long_short==1  ? longStop : na, style=plot.style_linebr, color=color.red, linewidth=1, title="Long Fixed SL")
plot(long_short==-1 ? shortStop : na, style=plot.style_linebr, color=color.red, linewidth=1, title="Short Fixed SL")
plot(long_short==1  ? longTake : na, style=plot.style_linebr, color=color.navy, linewidth=1, title="Long Fixed TP")
plot(long_short==-1 ? shortTake : na, style=plot.style_linebr, color=color.navy, linewidth=1, title="Short Fixed TP")
 
 
longBar1 = ta.barssince(long_last)
longBar2 = longBar1 >= 0 ? true : false
shortBar1 = ta.barssince(short_last)
shortBar2 = shortBar1 >= 0 ? true : false
 
longSLhit = long_short==1 and longBar2 and low < longStop
 
if(long_entered and not sec)
    longSLhit := true
if(long_entered and  sell_1)
    longSLhit := true 
plotshape(longSLhit and not (sell and  not(short_entered)  and (long_entered)), style=shape.labelup, location=location.belowbar, color=color.gray, size=size.tiny, title="Stop Loss", text="Long SL", textcolor=color.white)
shortSLhit = long_short==-1 and shortBar2 and high > shortStop
 
 
if(short_entered and not sec)
    shortSLhit := true
if(short_entered and buy_1)
    shortSLhit := true 
plotshape(shortSLhit and not (buy and  not(long_entered)  and (short_entered)), style=shape.labeldown, location=location.abovebar, color=color.gray, size=size.tiny, title="Stop Loss", text="Short SL", textcolor=color.white)
 
 
longTPhit = long_short==1 and longBar2 and high > longTake
plotshape(longTPhit, style=shape.labeldown, location=location.abovebar, color=color.navy, size=size.tiny, title="Target", text="Long TP", textcolor=color.white)
shortTPhit = long_short==-1 and shortBar2 and low < shortTake
plotshape(shortTPhit, style=shape.labelup, location=location.belowbar, color=color.navy, size=size.tiny, title="Target", text="Short TP", textcolor=color.white)
 
long_short := (long_short==1 or long_short==0) and longBar2 and (longSLhit or longTPhit) ? 0 : (long_short==-1 or long_short==0) and shortBar2 and (shortSLhit or shortTPhit) ? 0 : long_short
 
 
plotshape(buy and not(long_entered) , color = color.green , style = shape.labelup , text = "Buy" , textcolor = color.white , location = location.belowbar)
plotshape(sell and  not(short_entered)  , color = color.red , style = shape.labeldown , text = "Sell" , textcolor = color.white , location = location.abovebar)
 
 
// plotshape(buy and not(long_entered) and (short_entered), color = color.green , style = shape.labelup , text = "FA Buy" , textcolor = color.white , location = location.belowbar)
// plotshape(sell and not(short_entered)  and (long_entered), color = color.red , style = shape.labeldown , text = "FA Sell" , textcolor = color.white , location = location.abovebar)
 
 
// alertcondition(condition=buy and  not(long_entered)  and (short_entered), title="Fully Algo Buy")
// alertcondition(condition=sell and  not(short_entered)  and (long_entered), title="Fully Algo sell")
 
alertcondition(condition=buy  and not(long_entered), title="Buy")
alertcondition(condition=sell  and not(short_entered), title="Sell")
 
if(long_last)
    long_entered := true
    short_entered := false
if(short_last)
    short_entered := true
    long_entered := false
 
alertcondition(condition= (longSLhit or shortSLhit) and not (sell and  not(short_entered)  and (long_entered)), title="Long/Short SL")
//alertcondition(condition= shortSLhit and not (buy and  not(long_entered)  and (short_entered)), title="Short SL")
 
alertcondition(condition=longTPhit or shortTPhit, title="Long/short TP")
//alertcondition(condition=shortTPhit, title="Short TP")
 
if(longSLhit or longTPhit )
    long_entered := false
 
if(shortSLhit or shortTPhit)
    short_entered := false
